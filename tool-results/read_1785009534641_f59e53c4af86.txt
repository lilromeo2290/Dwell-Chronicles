   474→
   475→  // Tab state

   477→  const [tabCategory, setTabCategory] = useState('');
   478→  const [tabFeatured, setTabFeatured] = useState('');
   479→  const [tabNewlyAdded, setTabNewlyAdded] = useState('');
   480→  const [tabStatus, setTabStatus] = useState('');
   481→
   482→  // Dynamic filter options derived from apartment data
   483→  const [dynamicAreas, setDynamicAreas] = useState<{ label: string; value: string }[]>([
   484→    { label: 'All Areas', value: '' },
   485→  ]);
   486→  const [dynamicPriceRanges, setDynamicPriceRanges] = useState<
   487→    { label: string; value: string; min: string; max: string }[]
   488→  >([{ label: 'Any Price', value: '', min: '', max: '' }]);
   489→
   490→  // Data state
   491→  const [apartments, setApartments] = useState<Apartment[]>([]);
   492→  const [total, setTotal] = useState(0);
   493→  const [loading, setLoading] = useState(true);
   494→  const [loadingMore, setLoadingMore] = useState(false);
   495→
   496→  // Track if initial load
   497→  const [initialLoad, setInitialLoad] = useState(true);
   498→
   499→  // Parse price range value from combined select value
   500→  const parsePriceRange = useCallback((val: string) => {
   501→    if (!val) return { min: '', max: '' };
   502→    const parts = val.split('|');
   503→    return { min: parts[1] || '', max: parts[2] || '' };
   504→  }, []);
   505→
   506→  // Fetch apartments
   507→  const fetchApartments = useCallback(
   508→    async (offset = 0, append = false) => {
   509→      if (offset === 0) {
   510→        setLoading(true);
   511→      } else {
   512→        setLoadingMore(true);
   513→      }
   514→
   515→      try {
   516→        const params = new URLSearchParams();
   517→        if (search) params.set('search', search);
   518→        if (city) params.set('city', city);
   519→        if (area) params.set('area', area);
   520→        if (tabCategory) params.set('category', tabCategory);
   521→        if (tabFeatured) params.set('featured', tabFeatured);
   522→        if (tabNewlyAdded) params.set('newlyAdded', tabNewlyAdded);
   523→        if (tabStatus) params.set('status', tabStatus);
   524→
   525→        const { min, max } = parsePriceRange(priceRange);
   526→        if (min) params.set('minPrice', min);
   527→        if (max) params.set('maxPrice', max);
   528→
   529→        if (bedrooms) params.set('bedrooms', bedrooms);
   530→        if (guests) params.set('minGuests', guests);
   531→
   532→        params.set('limit', '20');
   533→        params.set('offset', String(offset));
   534→
   535→        const res = await fetch(`/api/apartments?${params.toString()}`);
   536→        const data = await res.json();
