   235→/*  Admin Password Gate                                                   */
   236→/* ------------------------------------------------------------------ */
   237→
   238→const ADMIN_CREDS = { username: 'airbnb', password: 'Airbnb/Clipeconsult' };
   239→
   240→function AdminPasswordGate({
   241→  isScrolled,
   242→  onSuccess,
   243→  onCancel,
   244→}: {
   245→  isScrolled: boolean;
   246→  onSuccess: () => void;
   247→  onCancel: () => void;
   248→}) {
   249→  const [username, setUsername] = useState('');
   250→  const [password, setPassword] = useState('');
   251→  const [error, setError] = useState('');
   252→  const inputRef = useRef<HTMLInputElement>(null);
   253→
   254→  useEffect(() => {
   255→    inputRef.current?.focus();
   256→  }, []);
   257→
   258→  const handleSubmit = (e: React.FormEvent) => {
   259→    e.preventDefault();
   260→    if (username === ADMIN_CREDS.username && password === ADMIN_CREDS.password) {
   261→      sessionStorage.setItem('dc_admin_auth', '1');
   262→      onSuccess();
   263→    } else {
   264→      setError('Invalid credentials');
   265→      setPassword('');
   266→    }
   267→  };
   268→
   269→  return (
   270→    <div className="p-3">
   271→      <p className={cn('text-xs font-medium mb-3', isScrolled ? 'text-white/70' : 'text-gray-500')}>
   272→        Enter admin credentials
   273→      </p>
   274→      <form onSubmit={handleSubmit} className="space-y-2">
   275→        <input
   276→          ref={inputRef}
   277→          type="text"
   278→          placeholder="Username"
   279→          value={username}
   280→          onChange={(e) => { setUsername(e.target.value); setError(''); }}
   281→          className={cn(
   282→            'w-full rounded-lg border px-3 py-1.5 text-xs outline-none transition-colors',
   283→            isScrolled
   284→              ? 'bg-white/10 border-white/20 text-white placeholder:text-white/40 focus:border-white/50'
   285→              : 'bg-gray-50 border-gray-200 text-gray-900 placeholder:text-gray-400 focus:border-[#5F8768]'
   286→          )}
   287→        />
   288→        <input
   289→          type="password"
   290→          placeholder="Password"
   291→          value={password}
   292→          onChange={(e) => { setPassword(e.target.value); setError(''); }}
   293→          className={cn(
   294→            'w-full rounded-lg border px-3 py-1.5 text-xs outline-none transition-colors',
   295→            isScrolled
   296→              ? 'bg-white/10 border-white/20 text-white placeholder:text-white/40 focus:border-white/50'
   297→              : 'bg-gray-50 border-gray-200 text-gray-900 placeholder:text-gray-400 focus:border-[#5F8768]'
   298→          )}
   299→        />
   300→        {error && (
   301→          <p className="text-xs text-red-400">{error}</p>
   302→        )}
   303→        <div className="flex gap-2 pt-1">
   304→          <button
   305→            type="submit"
   306→            className={cn(
   307→              'flex-1 rounded-lg px-3 py-1.5 text-xs font-medium transition-colors',
   308→              isScrolled
   309→                ? 'bg-white text-[#5F8768] hover:bg-white/90'
   310→                : 'bg-[#5F8768] text-white hover:bg-[#5F8768]/90'
   311→            )}
   312→          >
   313→            Sign In
   314→          </button>
   315→          <button
   316→            type="button"
   317→            onClick={onCancel}
   318→            className={cn(
   319→              'rounded-lg px-3 py-1.5 text-xs font-medium transition-colors',
   320→              isScrolled
   321→                ? 'text-white/70 hover:text-white'
   322→                : 'text-gray-500 hover:text-gray-700'
   323→            )}
   324→          >
   325→            Cancel
   326→          </button>
   327→        </div>
   328→      </form>
   329→    </div>
   330→  );
   331→}

   571→            {link.icon}

   573→          {link.label}
   574→        </Link>
   575→      ) : link.action ? (
   576→        <button

   578→          onClick={() => onAction(link.action!)}

   585→        </button>
   586→      ) : (
   587→        <a

   590→          className="group flex items-center gap-3.5 rounded-lg px-3 py-3 text-[15px] font-medium text-[#2F3A33] transition-colors duration-200 hover:bg-[#5F8768]/10 hover:text-[#5F8768]"

   595→          {link.label}
   596→        </a>
   597→      )}
   598→    </SheetClose>

   625→
   626→  const handleAdminClick = useCallback((href: string) => {
   627→    if (adminAuth) {
   628→      return; // already authenticated, link will navigate normally

   631→    setAdminUsername('');
   632→    setAdminPassword('');

   640→      sessionStorage.setItem('dc_admin_auth', '1');
