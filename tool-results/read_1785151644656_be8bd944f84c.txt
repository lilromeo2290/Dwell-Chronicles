   235→   235→/*  Admin Password Gate                                                   */
   236→   236→/* ------------------------------------------------------------------ */
   237→   237→
   238→   238→const ADMIN_CREDS = { username: 'airbnb', password: 'Airbnb/Clipeconsult' };
   239→   239→
   240→   240→function AdminPasswordGate({
   241→   241→  isScrolled,
   242→   242→  onSuccess,
   243→   243→  onCancel,
   244→   244→}: {
   245→   245→  isScrolled: boolean;
   246→   246→  onSuccess: () => void;
   247→   247→  onCancel: () => void;
   248→   248→}) {
   249→   249→  const [username, setUsername] = useState('');
   250→   250→  const [password, setPassword] = useState('');
   251→   251→  const [error, setError] = useState('');
   252→   252→  const inputRef = useRef<HTMLInputElement>(null);
   253→   253→
   254→   254→  useEffect(() => {
   255→   255→    inputRef.current?.focus();
   256→   256→  }, []);
   257→   257→
   258→   258→  const handleSubmit = (e: React.FormEvent) => {
   259→   259→    e.preventDefault();
   260→   260→    if (username === ADMIN_CREDS.username && password === ADMIN_CREDS.password) {
   261→   261→      sessionStorage.setItem('dc_admin_auth', '1');
   262→   262→      onSuccess();
   263→   263→    } else {
   264→   264→      setError('Invalid credentials');
   265→   265→      setPassword('');
   266→   266→    }
   267→   267→  };
   268→   268→
   269→   269→  return (
   270→   270→    <div className="p-3">
   271→   271→      <p className={cn('text-xs font-medium mb-3', isScrolled ? 'text-white/70' : 'text-gray-500')}>
   272→   272→        Enter admin credentials
   273→   273→      </p>
   274→   274→      <form onSubmit={handleSubmit} className="space-y-2">
   275→   275→        <input
   276→   276→          ref={inputRef}
   277→   277→          type="text"
   278→   278→          placeholder="Username"
   279→   279→          value={username}
   280→   280→          onChange={(e) => { setUsername(e.target.value); setError(''); }}
   281→   281→          className={cn(
   282→   282→            'w-full rounded-lg border px-3 py-1.5 text-xs outline-none transition-colors',
   283→   283→            isScrolled
   284→   284→              ? 'bg-white/10 border-white/20 text-white placeholder:text-white/40 focus:border-white/50'
   285→   285→              : 'bg-gray-50 border-gray-200 text-gray-900 placeholder:text-gray-400 focus:border-[#5F8768]'
   286→   286→          )}
   287→   287→        />
   288→   288→        <input
   289→   289→          type="password"
   290→   290→          placeholder="Password"
   291→   291→          value={password}
   292→   292→          onChange={(e) => { setPassword(e.target.value); setError(''); }}
   293→   293→          className={cn(
   294→   294→            'w-full rounded-lg border px-3 py-1.5 text-xs outline-none transition-colors',
   295→   295→            isScrolled
   296→   296→              ? 'bg-white/10 border-white/20 text-white placeholder:text-white/40 focus:border-white/50'
   297→   297→              : 'bg-gray-50 border-gray-200 text-gray-900 placeholder:text-gray-400 focus:border-[#5F8768]'
   298→   298→          )}
   299→   299→        />
   300→   300→        {error && (
   301→   301→          <p className="text-xs text-red-400">{error}</p>
   302→   302→        )}
   303→   303→        <div className="flex gap-2 pt-1">
   304→   304→          <button
   305→   305→            type="submit"
   306→   306→            className={cn(
   307→   307→              'flex-1 rounded-lg px-3 py-1.5 text-xs font-medium transition-colors',
   308→   308→              isScrolled
   309→   309→                ? 'bg-white text-[#5F8768] hover:bg-white/90'
   310→   310→                : 'bg-[#5F8768] text-white hover:bg-[#5F8768]/90'
   311→   311→            )}
   312→   312→          >
   313→   313→            Sign In
   314→   314→          </button>
   315→   315→          <button
   316→   316→            type="button"
   317→   317→            onClick={onCancel}
   318→   318→            className={cn(
   319→   319→              'rounded-lg px-3 py-1.5 text-xs font-medium transition-colors',
   320→   320→              isScrolled
   321→   321→                ? 'text-white/70 hover:text-white'
   322→   322→                : 'text-gray-500 hover:text-gray-700'
   323→   323→            )}
   324→   324→          >
   325→   325→            Cancel
   326→   326→          </button>
   327→   327→        </div>
   328→   328→      </form>
   329→   329→    </div>
   330→   330→  );
   331→   331→}
