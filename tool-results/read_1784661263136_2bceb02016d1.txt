   134→// ── Helpers ────────────────────────────────────────────────────────
   135→function parsePrice(str: string): number {
   136→  const match = str.match(/[\d,]+/)
   137→  if (!match) return 0
   138→  return parseInt(match[0].replace(/,/g, ''), 10)
   139→}
   140→
   141→function formatPrice(n: number): string {
   142→  return `GH₵ ${n.toLocaleString()}/month`
   143→}

   149→    case 'House':
   150→      return 'bg-amber-100 text-amber-800 border-amber-200'
   151→    case 'Flat':
   152→      return 'bg-sky-100 text-sky-800 border-sky-200'
   153→    case 'Studio':

   178→// Page Component
   179→// ══════════════════════════════════════════════════════════════════
   180→export default function AdminRentalsPage() {
   181→  const { toast } = useToast()
   182→
   183→  // ── State ──────────────────────────────────────────────────────
   184→  const [properties, setProperties] = useState<RentalProperty[]>(SAMPLE_DATA)
   185→  const [searchQuery, setSearchQuery] = useState('')
   186→  const [filterLocation, setFilterLocation] = useState('Any Location')
   187→  const [filterType, setFilterType] = useState('All Types')
   188→  const [filterPrice, setFilterPrice] = useState('Any Price')
   189→  const [filterBedrooms, setFilterBedrooms] = useState('Any')
   190→
   191→  const [sheetOpen, setSheetOpen] = useState(false)
   192→  const [editingId, setEditingId] = useState<string | null>(null)
   193→  const [formData, setFormData] = useState<FormData>(initialFormData)
   194→  const [formErrors, setFormErrors] = useState<Partial<Record<keyof FormData, string>>>({})
   195→
   196→  // ── Derived data ───────────────────────────────────────────────
   197→  const priceRange = useMemo(
   198→    () => PRICE_RANGES.find((r) => r.label === filterPrice) ?? PRICE_RANGES[0],
   199→    [filterPrice],
   200→  )
   201→
   202→  const filteredProperties = useMemo(() => {
   203→    return properties.filter((p) => {
   204→      // Search
   205→      if (searchQuery) {
   206→        const q = searchQuery.toLowerCase()
   207→        if (
   208→          !p.name.toLowerCase().includes(q) &&
   209→          !p.location.toLowerCase().includes(q) &&
   210→          !p.type.toLowerCase().includes(q)
   211→        ) {
   212→          return false
   213→        }
   214→      }
   215→      // Location
   216→      if (filterLocation !== 'Any Location' && p.location !== filterLocation) return false
   217→      // Type
   218→      if (filterType !== 'All Types' && p.type !== filterType) return false
   219→      // Price
   220→      if (p.priceNum < priceRange.min || p.priceNum > priceRange.max) return false
   221→      // Bedrooms
   222→      if (filterBedrooms !== 'Any') {
   223→        if (filterBedrooms === '4+') {
   224→          if (p.bedrooms < 4) return false
   225→        } else if (p.bedrooms !== parseInt(filterBedrooms, 10)) return false
   226→      }
   227→      return true
   228→    })
   229→  }, [properties, searchQuery, filterLocation, filterType, filterPrice, filterBedrooms, priceRange])
   230→
   231→  const stats = useMemo(
   232→    () => ({
   233→      total: properties.length,
   234→      available: properties.filter((p) => p.status === 'available').length,
   235→      taken: properties.filter((p) => p.status === 'taken').length,
   236→    }),
   237→    [properties],
   238→  )
   239→
   240→  // ── Actions ────────────────────────────────────────────────────
   241→  const toggleStatus = useCallback(
   242→    (id: string) => {
   243→      setProperties((prev) =>
   244→        prev.map((p) => {
   245→          if (p.id !== id) return p
   246→          const next = p.status === 'available' ? 'taken' : 'available'
   247→          toast({
   248→            title: `Status updated`,
   249→            description: `"${p.name}" is now ${next}.`,
   250→          })
   251→          return { ...p, status: next }
   252→        }),
   253→      )
   254→    },
   255→    [toast],
   256→  )
   257→
   258→  const deleteProperty = useCallback(
   259→    (id: string) => {
   260→      const p = properties.find((x) => x.id === id)
   261→      setProperties((prev) => prev.filter((x) => x.id !== id))
   262→      toast({
   263→        title: 'Property deleted',
   264→        description: p ? `"${p.name}" has been removed.` : 'Property has been removed.',
   265→        variant: 'destructive',
   266→      })
   267→    },
   268→    [properties, toast],
   269→  )
   270→
   271→  const openAddSheet = () => {
   272→    setEditingId(null)
   273→    setFormData(initialFormData)
   274→    setFormErrors({})
   275→    setSheetOpen(true)
   276→  }
   277→
   278→  const openEditSheet = (p: RentalProperty) => {
   279→    setEditingId(p.id)
   280→    setFormData({
   281→      name: p.name,
   282→      location: p.location,
   283→      type: p.type,
   284→      price: String(p.priceNum),
   285→      bedrooms: String(p.bedrooms),
   286→      status: p.status,
   287→    })
   288→    setFormErrors({})
   289→    setSheetOpen(true)
   290→  }
   291→
   292→  const validateForm = (): boolean => {
   293→    const errors: Partial<Record<keyof FormData, string>> = {}
   294→    if (!formData.name.trim()) errors.name = 'Name is required'
   295→    if (!formData.location) errors.location = 'Location is required'
   296→    if (!formData.type) errors.type = 'Property type is required'
   297→    const priceNum = parseInt(formData.price, 10)
   298→    if (!formData.price || isNaN(priceNum) || priceNum <= 0) errors.price = 'Enter a valid price'
   299→    if (!formData.bedrooms || parseInt(formData.bedrooms, 10) < 1) errors.bedrooms = 'Bedrooms required'
   300→    setFormErrors(errors)
   301→    return Object.keys(errors).length === 0
   302→  }
   303→
   304→  const handleSave = () => {
   305→    if (!validateForm()) return
   306→    const priceNum = parseInt(formData.price, 10)
   307→
   308→    if (editingId) {
   309→      setProperties((prev) =>
   310→        prev.map((p) =>
   311→          p.id === editingId
   312→            ? {
   313→                ...p,
   314→                name: formData.name.trim(),
   315→                location: formData.location,
   316→                type: formData.type,
   317→                price: formatPrice(priceNum),
   318→                priceNum,
   319→                bedrooms: parseInt(formData.bedrooms, 10),
   320→                status: formData.status,
   321→              }
   322→            : p,
   323→        ),
   324→      )
   325→      toast({ title: 'Property updated', description: `"${formData.name}" has been updated.` })
   326→    } else {
   327→      const newProp: RentalProperty = {
   328→        id: String(Date.now()),
   329→        name: formData.name.trim(),
   330→        location: formData.location,
   331→        type: formData.type,
   332→        price: formatPrice(priceNum),
   333→        priceNum,
   334→        bedrooms: parseInt(formData.bedrooms, 10),
   335→        status: formData.status,
   336→      }
   337→      setProperties((prev) => [...prev, newProp])
   338→      toast({ title: 'Property added', description: `"${formData.name}" has been added.` })
   339→    }
   340→    setSheetOpen(false)
   341→  }
   342→
   343→  const updateField = <K extends keyof FormData>(key: K, value: FormData[K]) => {
   344→    setFormData((prev) => ({ ...prev, [key]: value }))
   345→    setFormErrors((prev) => {
   346→      if (!prev[key]) return prev
   347→      const next = { ...prev }
   348→      delete next[key]
   349→      return next
   350→    })
   351→  }
   352→
   353→  // ── Render ─────────────────────────────────────────────────────
   354→  return (
   355→    <div
   356→      className="min-h-screen"
   357→      style={{ backgroundColor: BRAND.offWhite }}
   358→    >
   359→      <main className="mx-auto max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   360→        {/* ─── Top Bar ─── */}
   361→        <header className="mb-8 flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   362→          <div>
   363→            <Link
   364→              href="/"
   365→              className="mb-2 inline-flex items-center gap-1.5 text-sm font-medium transition-colors hover:text-[#5F8768]"
   366→              style={{ color: BRAND.charcoal }}
   367→            >
   368→              <ArrowLeft className="size-4" />
   369→              Back to Site
   370→            </Link>
   371→            <h1
   372→              className="text-2xl font-bold tracking-tight sm:text-3xl"
   373→              style={{ color: BRAND.charcoal }}
   374→            >
   375→              Rental Properties Management
   376→            </h1>
   377→            <p className="mt-1 text-sm text-muted-foreground">
   378→              Manage rental property listings and their availability status
   379→            </p>
   380→          </div>
   381→          <Button
   382→            onClick={openAddSheet}
   383→            className="mt-3 shrink-0 sm:mt-0"
   384→            style={{ backgroundColor: BRAND.sage }}
   385→          >
   386→            <Plus className="size-4" />
   387→            Add New Property
   388→          </Button>
   389→        </header>
   390→
   391→        {/* ─── Filter Section ─── */}
   392→        <section
   393→          className="mb-6 rounded-xl border p-4 sm:p-6"
   394→          style={{ backgroundColor: '#fff', borderColor: BRAND.stone }}
   395→        >
   396→          <div className="mb-4 flex items-center gap-2">
   397→            <Search className="size-4 text-muted-foreground" />
   398→            <span className="text-sm font-semibold" style={{ color: BRAND.charcoal }}>
   399→              Filter Properties
   400→            </span>
   401→          </div>
   402→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-5">
   403→            {/* Search */}
   404→            <div className="relative lg:col-span-1">
   405→              <Search className="absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   406→              <Input
   407→                placeholder="Search properties..."
   408→                value={searchQuery}
   409→                onChange={(e) => setSearchQuery(e.target.value)}
   410→                className="pl-9"
   411→              />
   412→            </div>
   413→            {/* Location */}
   414→            <Select value={filterLocation} onValueChange={setFilterLocation}>
   415→              <SelectTrigger className="w-full">
   416→                <SelectValue placeholder="Location" />
   417→              </SelectTrigger>
   418→              <SelectContent>
   419→                {LOCATIONS.map((loc) => (
   420→                  <SelectItem key={loc} value={loc}>
   421→                    {loc}
   422→                  </SelectItem>
   423→                ))}
   424→              </SelectContent>
   425→            </Select>
   426→            {/* Type */}
   427→            <Select value={filterType} onValueChange={setFilterType}>
   428→              <SelectTrigger className="w-full">
   429→                <SelectValue placeholder="Property Type" />
   430→              </SelectTrigger>
   431→              <SelectContent>
   432→                {PROPERTY_TYPES.map((t) => (
   433→                  <SelectItem key={t} value={t}>
   434→                    {t}
   435→                  </SelectItem>
   436→                ))}
   437→              </SelectContent>
   438→            </Select>
   439→            {/* Price Range */}
   440→            <Select value={filterPrice} onValueChange={setFilterPrice}>
   441→              <SelectTrigger className="w-full">
   442→                <SelectValue placeholder="Price Range" />
   443→              </SelectTrigger>
   444→              <SelectContent>
   445→                {PRICE_RANGES.map((r) => (
   446→                  <SelectItem key={r.label} value={r.label}>
   447→                    {r.label}
   448→                  </SelectItem>
   449→                ))}
   450→              </SelectContent>
   451→            </Select>
   452→            {/* Bedrooms */}
   453→            <Select value={filterBedrooms} onValueChange={setFilterBedrooms}>
   454→              <SelectTrigger className="w-full">
   455→                <SelectValue placeholder="Bedrooms" />
   456→              </SelectTrigger>
   457→              <SelectContent>
   458→                {BEDROOM_OPTIONS.map((b) => (
   459→                  <SelectItem key={b} value={b}>
   460→                    {b}
   461→                  </SelectItem>
   462→                ))}
   463→              </SelectContent>
   464→            </Select>
   465→          </div>
   466→        </section>
   467→
   468→        {/* ─── Stats Row ─── */}
   469→        <section className="mb-6 grid grid-cols-1 gap-4 sm:grid-cols-3">
   470→          {[
   471→            { label: 'Total Properties', value: stats.total, icon: Building2, color: BRAND.charcoal, bg: BRAND.stone },
   472→            { label: 'Available', value: stats.available, icon: Home, color: '#15803d', bg: '#dcfce7' },
   473→            { label: 'Taken', value: stats.taken, icon: MapPin, color: '#b91c1c', bg: '#fef2f2' },
   474→          ].map((s) => (
   475→            <motion.div
   476→              key={s.label}
   477→              initial={{ opacity: 0, y: 12 }}
   478→              animate={{ opacity: 1, y: 0 }}
   479→              transition={{ duration: 0.3 }}
   480→              className="flex items-center gap-4 rounded-xl border p-4 sm:p-5"
   481→              style={{ backgroundColor: '#fff', borderColor: BRAND.stone }}
   482→            >

   484→                className="flex size-12 shrink-0 items-center justify-center rounded-lg"
   485→                style={{ backgroundColor: s.bg }}

   487→                <s.icon className="size-5" style={{ color: s.color }} />

   490→                <p className="text-sm text-muted-foreground">{s.label}</p>
   491→                <p className="text-2xl font-bold" style={{ color: s.color }}>

   493→                </p>
   494→              </div>
   495→            </motion.div>
   496→          ))}
   497→        </section>
   498→
   499→        {/* ─── Properties (Desktop Table) ─── */}
   500→        <section
   501→          className="hidden rounded-xl border md:block"
   502→          style={{ backgroundColor: '#fff', borderColor: BRAND.stone }}
   503→        >
   504→          <Table>
   505→            <TableHeader>
   506→              <TableRow style={{ borderBottomColor: BRAND.stone }}>
   507→                <TableHead className="font-semibold" style={{ color: BRAND.charcoal }}>Property</TableHead>
   508→                <TableHead className="font-semibold" style={{ color: BRAND.charcoal }}>Location</TableHead>
   509→                <TableHead className="font-semibold" style={{ color: BRAND.charcoal }}>Type</TableHead>
   510→                <TableHead className="font-semibold" style={{ color: BRAND.charcoal }}>Price</TableHead>
   511→                <TableHead className="font-semibold" style={{ color: BRAND.charcoal }}>Beds</TableHead>
   512→                <TableHead className="font-semibold" style={{ color: BRAND.charcoal }}>Status</TableHead>
   513→                <TableHead className="text-right font-semibold" style={{ color: BRAND.charcoal }}>Actions</TableHead>
   514→              </TableRow>
   515→            </TableHeader>
   516→            <TableBody>
   517→              <AnimatePresence mode="popLayout">
   518→                {filteredProperties.map((p, i) => (
   519→                  <motion.tr
   520→                    key={p.id}
   521→                    custom={i}
   522→                    variants={cardVariants}
   523→                    initial="hidden"
   524→                    animate="visible"
   525→                    exit="exit"
   526→                    className="border-b transition-colors hover:bg-muted/50"
   527→                    style={{ borderBottomColor: BRAND.stone }}
   528→                  >
   529→                    <TableCell className="font-medium" style={{ color: BRAND.charcoal }}>
   530→                      <div className="flex items-center gap-2">
   531→                        <Home className="size-4 shrink-0 text-muted-foreground" />
   532→                        {p.name}
   533→                      </div>
   534→                    </TableCell>
   535→                    <TableCell>
   536→                      <div className="flex items-center gap-1.5 text-sm text-muted-foreground">
   537→                        <MapPin className="size-3.5 shrink-0" />
   538→                        {p.location}
   539→                      </div>
   540→                    </TableCell>
   541→                    <TableCell>
   542→                      <Badge variant="outline" className={getTypeBadgeColor(p.type)}>

   546→                    </TableCell>
   547→                    <TableCell className="font-semibold" style={{ color: BRAND.charcoal }}>
   548→                      {p.price}
   549→                    </TableCell>
