   135→// ── Helpers ────────────────────────────────────────────────────────
   136→function parsePrice(str: string): number {
   137→  const match = str.match(/[\d,]+/)
   138→  if (!match) return 0
   139→  return parseInt(match[0].replace(/,/g, ''), 10)
   140→}
   141→
   142→function formatPrice(n: number): string {
   143→  return `GH₵ ${n.toLocaleString()}/month`
   144→}
