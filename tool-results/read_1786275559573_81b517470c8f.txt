   312→function CustomCalendar({
   313→  availabilityDates,
   314→  checkIn,
   315→  checkOut,
   316→  onSelectDate,
   317→  onMonthChange,
   318→}: {
   319→  availabilityDates: AvailabilityDate[];
   320→  checkIn: Date | null;
   321→  checkOut: Date | null;
   322→  onSelectDate: (date: Date) => void;
   323→  onMonthChange: (monthStr: string) => void;
   324→}) {
   325→  const [currentMonth, setCurrentMonth] = useState(startOfMonth(new Date()));
   326→  const isMobile = useIsMobile();
   327→
   328→  // Notify parent of month changes for data fetching
   329→  const handleMonthChange = (newMonth: Date) => {
   330→    setCurrentMonth(newMonth);
   331→    onMonthChange(format(newMonth, 'yyyy-MM'));
   332→  };
   333→
   334→  const prevMonth = () => handleMonthChange(subMonths(currentMonth, 1));
   335→  const nextMonth = () => handleMonthChange(addMonths(currentMonth, 1));
   336→
   337→  const availabilityMap = useMemo(() => {
   338→    const map = new Map<string, string>();
   339→    availabilityDates.forEach((d) => map.set(d.date, d.status));
   340→    return map;
   341→  }, [availabilityDates]);
   342→
   343→  const daysInMonth = eachDayOfInterval({
   344→    start: startOfMonth(currentMonth),
   345→    end: endOfMonth(currentMonth),
   346→  });
   347→
   348→  const firstDayOffset = getDayOfWeekMondayBased(startOfMonth(currentMonth));
   349→
   350→
   351→
   352→  const isDateInRange = (date: Date) => {
   353→    if (!checkIn || !checkOut) return false;
   354→    const d = startOfDay(date);
   355→    const ci = startOfDay(checkIn);
   356→    const co = startOfDay(checkOut);
   357→    return (d > ci && d < co);
   358→  };
   359→
   360→  const isCheckInDate = (date: Date) => {
   361→    if (!checkIn) return false;
   362→    return date.toDateString() === checkIn.toDateString();
   363→  };
   364→
   365→  const isCheckOutDate = (date: Date) => {
   366→    if (!checkOut) return false;
   367→    return date.toDateString() === checkOut.toDateString();
   368→  };
   369→
   370→  const handleDateClick = (date: Date) => {
   371→    const dateStr = format(date, 'yyyy-MM-dd');
   372→    const status = availabilityMap.get(dateStr) || 'available';
   373→    if (status !== 'available') return;
   374→    if (isBefore(startOfDay(date), startOfDay(new Date()))) return;
   375→    onSelectDate(date);
   376→  };
   377→
   378→  const isDateDisabled = (date: Date) => {
   379→    const dateStr = format(date, 'yyyy-MM-dd');
   380→    const status = availabilityMap.get(dateStr) || 'available';
   381→    return status !== 'available' || isBefore(startOfDay(date), startOfDay(new Date()));
   382→  };
   383→
   384→  return (
   385→    <div className="w-full">
   386→      {/* Calendar Header */}
   387→      <div className="flex items-center justify-between mb-4">
   388→        <button
   389→          onClick={prevMonth}
   390→          className="p-2 rounded-lg hover:bg-[#D8D5CC]/50 transition-colors"
   391→          aria-label="Previous month"
   392→        >
   393→          <ChevronLeft className="w-5 h-5 text-[#2F3A33]" />
   394→        </button>
   395→        <h3 className="text-lg font-semibold text-[#2F3A33]">
   396→          {MONTH_NAMES[currentMonth.getMonth()]} {currentMonth.getFullYear()}
   397→        </h3>
   398→        <button
   399→          onClick={nextMonth}
   400→          className="p-2 rounded-lg hover:bg-[#D8D5CC]/50 transition-colors"
   401→          aria-label="Next month"
   402→        >
   403→          <ChevronRight className="w-5 h-5 text-[#2F3A33]" />
   404→        </button>
   405→      </div>
   406→
   407→      {/* Day Names */}
   408→      <div className="grid grid-cols-7 mb-1">
   409→        {DAY_NAMES.map((day) => (
   410→          <div
   411→            key={day}
   412→            className="text-center text-xs font-medium text-gray-400 py-2"
   413→          >
   414→            {isMobile ? day.charAt(0) : day}
   415→          </div>
   416→        ))}
   417→      </div>
   418→
   419→      {/* Date Cells */}
   420→      <div className="grid grid-cols-7 gap-0.5">
   421→        {/* Empty cells for offset */}
   422→        {Array.from({ length: firstDayOffset }).map((_, i) => (
   423→          <div key={`empty-${i}`} className="aspect-square" />
   424→        ))}
   425→
   426→        {/* Date cells */}
   427→        {daysInMonth.map((date) => {
   428→          const dateStr = format(date, 'yyyy-MM-dd');
   429→          const status = availabilityMap.get(dateStr) || 'available';
   430→          const disabled = isDateDisabled(date);
   431→          const inRange = isDateInRange(date);
   432→          const isCI = isCheckInDate(date);
   433→          const isCO = isCheckOutDate(date);
   434→          const today = isToday(date);
   435→
   436→          return (
   437→            <button
   438→              key={dateStr}
   439→              onClick={() => handleDateClick(date)}
   440→              disabled={disabled}
   441→              className={`
   442→                relative aspect-square flex items-center justify-center text-sm rounded-lg
   443→                transition-all duration-150 font-medium
   444→                ${disabled ? 'cursor-not-allowed opacity-40' : 'cursor-pointer'}
   445→                ${isCI ? 'bg-[#5F8768] text-white rounded-lg font-bold shadow-md' : ''}
   446→                ${isCO ? 'bg-[#5F8768] text-white rounded-lg font-bold shadow-md' : ''}
   447→                ${inRange && !isCI && !isCO ? 'bg-[#5F8768]/15 text-[#2F3A33]' : ''}
   448→                ${!disabled && !isCI && !isCO && !inRange ? STATUS_CALENDAR_COLORS[status] || 'bg-white' : ''}
   449→                ${today && !isCI && !isCO ? 'ring-2 ring-[#5F8768] ring-offset-1' : ''}
   450→                hover:scale-105
   451→              `}
   452→              aria-label={`${format(date, 'MMM d, yyyy')} - ${status}`}
   453→            >
   454→              {format(date, 'd')}
   455→              {status === 'booked' && !isCI && !isCO && (
   456→                <span className="absolute text-[8px] font-bold text-red-500">×</span>
   457→              )}
   458→            </button>
   459→          );
   460→        })}
   461→      </div>
   462→
   463→      {/* Legend */}
   464→      <div className="flex flex-wrap gap-4 mt-4 text-xs text-gray-500">
   465→        <div className="flex items-center gap-1.5">
   466→          <div className="w-3 h-3 rounded bg-emerald-100 border border-emerald-200" />
   467→          Available
   468→        </div>
   469→        <div className="flex items-center gap-1.5">
   470→          <div className="w-3 h-3 rounded bg-red-100 border border-red-200" />
   471→          Booked
   472→        </div>
   473→        <div className="flex items-center gap-1.5">
   474→          <div className="w-3 h-3 rounded bg-amber-100 border border-amber-200" />
   475→          Reserved
   476→        </div>
   477→        <div className="flex items-center gap-1.5">
   478→          <div className="w-3 h-3 rounded bg-gray-100 border border-gray-200" />
   479→          Maintenance
   480→        </div>
   481→      </div>
   482→    </div>
   483→  );
   484→}

   488→function FullscreenGallery({
   489→  images,
   490→  currentIndex,
   491→  onClose,
   492→  onPrev,
   493→  onNext,
   494→}: {
   495→  images: ApartmentImage[];
   496→  currentIndex: number;
   497→  onClose: () => void;
   498→  onPrev: () => void;
   499→  onNext: () => void;
   500→}) {
   501→  const touchStartX = useRef(0);
   502→
   503→  useEffect(() => {
   504→    const handleKeyDown = (e: KeyboardEvent) => {
   505→      if (e.key === 'Escape') onClose();
   506→      if (e.key === 'ArrowLeft') onPrev();
   507→      if (e.key === 'ArrowRight') onNext();
   508→    };
   509→    document.addEventListener('keydown', handleKeyDown);
   510→    document.body.style.overflow = 'hidden';
   511→    return () => {
   512→      document.removeEventListener('keydown', handleKeyDown);
   513→      document.body.style.overflow = '';
   514→    };
   515→  }, [onClose, onPrev, onNext]);
   516→
   517→  const handleTouchStart = (e: React.TouchEvent) => {
   518→    touchStartX.current = e.touches[0].clientX;
   519→  };
   520→
   521→  const handleTouchEnd = (e: React.TouchEvent) => {
   522→    const diff = touchStartX.current - e.changedTouches[0].clientX;
   523→    if (Math.abs(diff) > 50) {
   524→      if (diff > 0) onNext();
   525→      else onPrev();
   526→    }
   527→  };
   528→
   529→  return (
   530→    <motion.div
   531→      initial={{ opacity: 0 }}
   532→      animate={{ opacity: 1 }}
   533→      exit={{ opacity: 0 }}
   534→      className="fixed inset-0 z-50 bg-black/95 flex items-center justify-center"
   535→      onClick={onClose}
   536→    >
   537→      {/* Close button */}
   538→      <button
   539→        onClick={onClose}
   540→        className="absolute top-4 right-4 z-10 p-2 rounded-full bg-white/10 hover:bg-white/20 transition-colors"
   541→        aria-label="Close gallery"
   542→      >
   543→        <X className="w-6 h-6 text-white" />
   544→      </button>
   545→
   546→      {/* Counter */}
   547→      <div className="absolute top-4 left-4 z-10 text-white/80 text-sm font-medium bg-black/40 px-3 py-1.5 rounded-full">
   548→        {currentIndex + 1} / {images.length}
   549→      </div>
   550→
   551→      {/* Navigation arrows - desktop */}
   552→      <button
   553→        onClick={(e) => { e.stopPropagation(); onPrev(); }}
   554→        className="hidden md:flex absolute left-4 top-1/2 -translate-y-1/2 z-10 p-3 rounded-full bg-white/10 hover:bg-white/20 transition-colors"
   555→        aria-label="Previous image"
   556→      >
   557→        <ChevronLeft className="w-6 h-6 text-white" />
   558→      </button>
   559→      <button
   560→        onClick={(e) => { e.stopPropagation(); onNext(); }}
   561→        className="hidden md:flex absolute right-4 top-1/2 -translate-y-1/2 z-10 p-3 rounded-full bg-white/10 hover:bg-white/20 transition-colors"
   562→        aria-label="Next image"
   563→      >
   564→        <ChevronRight className="w-6 h-6 text-white" />
   565→      </button>
   566→
   567→      {/* Image */}
   568→      <motion.div
   569→        key={currentIndex}
   570→        initial={{ opacity: 0, scale: 0.95 }}
   571→        animate={{ opacity: 1, scale: 1 }}
   572→        exit={{ opacity: 0, scale: 0.95 }}
   573→        transition={{ duration: 0.2 }}
   574→        className="w-full h-full flex items-center justify-center p-8 md:p-16"
   575→        onClick={(e) => e.stopPropagation()}
   576→        onTouchStart={handleTouchStart}
   577→        onTouchEnd={handleTouchEnd}
   578→      >
   579→        <img
   580→          src={images[currentIndex]?.url}
   581→          alt={images[currentIndex]?.alt || 'Apartment image'}
