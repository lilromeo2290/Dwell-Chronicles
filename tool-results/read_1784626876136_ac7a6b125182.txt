   263→function formatCurrency(amount: number | null | undefined): string {
   264→  if (amount === null || amount === undefined) return '—';
   265→  return `GH\u20B5${amount.toLocaleString()}`;
   266→}

   268→function formatDate(dateStr: string): string {
   269→  try {
   270→    return new Date(dateStr).toLocaleDateString('en-GB', {
   271→      day: '2-digit',
   272→      month: 'short',
   273→      year: 'numeric',
   274→    });
   275→  } catch {
   276→    return dateStr;
   277→  }
   278→}
   279→
   280→function formatDateTime(dateStr: string): string {
   281→  try {
   282→    return new Date(dateStr).toLocaleDateString('en-GB', {
   283→      day: '2-digit',
   284→      month: 'short',
   285→      year: 'numeric',
   286→      hour: '2-digit',
   287→      minute: '2-digit',
   288→    });
   289→  } catch {
   290→    return dateStr;
   291→  }
   292→}
   293→
   294→function getDaysInMonth(year: number, month: number): number {
