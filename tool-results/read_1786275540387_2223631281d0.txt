   141→function getYouTubeEmbedUrl(url?: string) {
   142→  if (!url) return null;
   143→  const match = url.match(/(?:youtube\.com\/(?:watch\?v=|embed\/)|youtu\.be\/)([\w-]+)/);
   144→  return match ? `https://www.youtube.com/embed/${match[1]}` : null;

   147→function MortgageEstimate({ priceNum, status }: { priceNum: number; status: string }) {
   148→  if (status === 'For Rent' || status === 'Rented' || status === 'Available') return null;
   149→  const downPayment = priceNum * 0.2;
   150→  const loanAmount = priceNum - downPayment;
   151→  const rate = 0.25 / 12;
   152→  const months = 20 * 12;
   153→  const monthly = (loanAmount * rate * Math.pow(1 + rate, months)) / (Math.pow(1 + rate, months) - 1);
   154→  const totalPayment = monthly * months;
   155→  return (
   156→    <div className="rounded-2xl border border-[#E5E3DC] bg-white p-6">
   157→      <div className="flex items-center gap-2 mb-4">
   158→        <Calculator className="h-5 w-5 text-[#5F8768]" />
   159→        <h3 className="font-semibold text-[#2F3A33]">Mortgage Estimate</h3>
   160→      </div>
   161→      <div className="grid grid-cols-2 gap-4 text-sm">
   162→        <div><p className="text-[#6B7A6F]">Property Price</p><p className="font-semibold text-[#2F3A33]">GH\u20b5 {priceNum.toLocaleString()}</p></div>
   163→        <div><p className="text-[#6B7A6F]">20% Down Payment</p><p className="font-semibold text-[#2F3A33]">GH\u20b5 {downPayment.toLocaleString()}</p></div>
   164→        <div><p className="text-[#6B7A6F]">Loan Amount</p><p className="font-semibold text-[#2F3A33]">GH\u20b5 {loanAmount.toLocaleString()}</p></div>
   165→        <div><p className="text-[#6B7A6F]">Interest Rate</p><p className="font-semibold text-[#2F3A33]">25% / year</p></div>
   166→        <div><p className="text-[#6B7A6F]">Loan Term</p><p className="font-semibold text-[#2F3A33]">20 years</p></div>
   167→        <div><p className="text-[#6B7A6F]">Monthly Payment</p><p className="font-bold text-[#5F8768] text-base">GH\u20b5 {Math.round(monthly).toLocaleString()}</p></div>
