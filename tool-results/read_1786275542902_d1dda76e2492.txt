   467→
   468→  // Filter state

   474→  const [guests, setGuests] = useState('');
   475→
   476→  // Tab state
   477→  const [activeTab, setActiveTab] = useState(0);
   478→  const [tabCategory, setTabCategory] = useState('');
   479→  const [tabFeatured, setTabFeatured] = useState('');
   480→  const [tabNewlyAdded, setTabNewlyAdded] = useState('');
   481→  const [tabStatus, setTabStatus] = useState('');
   482→
   483→  // Dynamic filter options derived from apartment data
   484→  const [dynamicAreas, setDynamicAreas] = useState<{ label: string; value: string }[]>([
   485→    { label: 'All Areas', value: '' },
   486→  ]);
   487→  const [dynamicPriceRanges, setDynamicPriceRanges] = useState<
   488→    { label: string; value: string; min: string; max: string }[]
   489→  >([{ label: 'Any Price', value: '', min: '', max: '' }]);
   490→
   491→  // Data state
   492→  const [apartments, setApartments] = useState<Apartment[]>([]);
   493→  const [total, setTotal] = useState(0);
   494→  const [loading, setLoading] = useState(true);
   495→  const [loadingMore, setLoadingMore] = useState(false);
   496→
   497→  // Track if initial load
   498→  const [initialLoad, setInitialLoad] = useState(true);
   499→
   500→  // Parse price range value from combined select value
   501→  const parsePriceRange = useCallback((val: string) => {
   502→    if (!val) return { min: '', max: '' };
   503→    const parts = val.split('|');
   504→    return { min: parts[1] || '', max: parts[2] || '' };
   505→  }, []);
   506→
   507→  // Fetch apartments
   508→  const fetchApartments = useCallback(
   509→    async (offset = 0, append = false) => {
   510→      if (offset === 0) {
   511→        setLoading(true);
   512→      } else {
   513→        setLoadingMore(true);
   514→      }
   515→
   516→      try {
   517→        const params = new URLSearchParams();
   518→        if (search) params.set('search', search);
   519→        if (city) params.set('city', city);
   520→        if (area) params.set('area', area);
   521→        if (tabCategory) params.set('category', tabCategory);
   522→        if (tabFeatured) params.set('featured', tabFeatured);
   523→        if (tabNewlyAdded) params.set('newlyAdded', tabNewlyAdded);
   524→        if (tabStatus) params.set('status', tabStatus);
   525→
   526→        const { min, max } = parsePriceRange(priceRange);
   527→        if (min) params.set('minPrice', min);
   528→        if (max) params.set('maxPrice', max);
   529→
   530→        if (bedrooms) params.set('bedrooms', bedrooms);
   531→        if (guests) params.set('minGuests', guests);
   532→
   533→        params.set('limit', '20');
   534→        params.set('offset', String(offset));
   535→
   536→        const res = await fetch(`/api/apartments?${params.toString()}`);
   537→        const data = await res.json();
   538→
   539→        if (res.ok) {
   540→          const fetched: Apartment[] = data.apartments;
   541→          if (append) {
   542→            setApartments((prev) => [...prev, ...fetched]);
   543→          } else {
   544→            setApartments(fetched);
   545→          }
   546→          setTotal(data.total);
   547→
   548→          // Build dynamic filter options from apartment data (only on initial full load)
   549→          if (!append && initialLoad) {
   550→            // Derive unique areas
   551→            const allApts = data.apartments || [];
   552→            const uniqueAreas = [...new Set(allApts.map((a: Apartment) => a.area).filter(Boolean))].sort();
   553→            setDynamicAreas([
   554→              { label: 'All Areas', value: '' },
   555→              ...uniqueAreas.map((a: string) => ({ label: a, value: a })),
   556→            ]);
   557→
   558→            // Derive price ranges from actual prices
   559→            const prices = allApts.map((a: Apartment) => a.pricePerNight).filter((p: number) => p > 0);
   560→            if (prices.length > 0) {
   561→              const minP = Math.min(...prices);
   562→              const maxP = Math.max(...prices);
   563→              const ranges: { label: string; value: string; min: string; max: string }[] = [
   564→                { label: 'Any Price', value: '', min: '', max: '' },
   565→              ];
   566→              const brackets = [100, 200, 400, 600, 1000, 1500, 2000].filter(
   567→                (b) => b >= Math.floor(minP / 100) * 100
   568→              );
   569→              for (let i = 0; i < brackets.length; i++) {
   570→                const lo = i === 0 ? Math.floor(minP / 100) * 100 : brackets[i - 1];
   571→                const hi = brackets[i];
   572→                if (lo < hi) {
   573→                  ranges.push({
   574→                    label: `GH₵ ${lo.toLocaleString()} - GH₵ ${hi.toLocaleString()}`,
   575→                    value: `${lo}|${hi}`,
   576→                    min: String(lo),
   577→                    max: String(hi),
   578→                  });
   579→                }
   580→              }
   581→              // "Over" bracket for the highest
   582→              const lastBracket = brackets[brackets.length - 1];
   583→              if (lastBracket < maxP) {
   584→                ranges.push({
   585→                  label: `Over GH₵ ${lastBracket.toLocaleString()}`,
   586→                  value: `${lastBracket}|`,
   587→                  min: String(lastBracket),
   588→                  max: '',
   589→                });
   590→              }
   591→              setDynamicPriceRanges(ranges);
   592→            }
   593→          }
   594→        }
   595→      } catch (err) {
   596→        console.error('Failed to fetch apartments:', err);
   597→      } finally {
   598→        setLoading(false);
   599→        setLoadingMore(false);
   600→        setInitialLoad(false);
   601→      }
   602→    },
   603→    [search, city, area, tabCategory, tabFeatured, tabNewlyAdded, tabStatus, priceRange, bedrooms, guests, parsePriceRange]
   604→  );

   622→  const handleGuestsChange = (val: string) => {

   626→

   682→                'radial-gradient(circle at 20% 50%, rgba(255,255,255,0.15) 0%, transparent 50%), radial-gradient(circle at 80% 20%, rgba(255,255,255,0.1) 0%, transparent 50%)',

   684→          />
   685→        </div>

   687→        <div className="relative max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8 sm:py-10">
   688→          {/* Back Link */}
   689→          <Link

   699→              Short-Stay Apartments
   700→              <br />

   726→            </div>

   778→              <Input

   780→                value={search}
